// Firebase Firestore Security Rules for Safe Voice App
// Add these rules to your existing firestore.rules file

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    
    // ==================== APP USAGE ANALYTICS ====================
    // Allow admins to read analytics data
    // Allow system/Cloud Functions to write (if using direct writes, restrict in production)
    match /app_usage_analytics/{document=**} {
      // Allow read access to authenticated admin users
      // You should verify admin role in production
      allow read: if request.auth != null;
      
      // Allow write access (for now - move to Cloud Functions in production)
      // In production, set this to: allow write: if false;
      // and use Cloud Functions for all writes
      allow write: if true; // TEMPORARY - change to false in production
    }
    
    // ==================== REPORTS ====================
    // Your existing reports rules should remain unchanged
    match /reports/{reportId} {
      // Your existing report rules here
      allow read: if request.auth != null;
      allow create: if true; // Anonymous submissions
      allow update: if request.auth != null;
      allow delete: if request.auth != null;
    }
    
    // ==================== ADMIN NOTIFICATIONS ====================
    match /admin_notifications/{notificationId} {
      allow read: if request.auth != null;
      allow write: if request.auth != null;
    }
    
    // ==================== OTHER COLLECTIONS ====================
    // Add your other collection rules here
  }
}

// IMPORTANT NOTES:
// 1. The analytics write rule (allow write: if true) is temporary for development
// 2. In production, change to (allow write: if false) and use Cloud Functions
// 3. Add proper admin role verification instead of just checking request.auth != null
// 4. Example admin check: request.auth.token.admin == true
