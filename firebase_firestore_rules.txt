// Firebase Firestore Security Rules for Anonymous Reporting App
// These rules should be applied in the Firebase Console -> Firestore -> Rules

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    
    // Helper function to check if user is authenticated admin
    function isAdmin() {
      return request.auth != null && request.auth.token.email != null;
    }
    
    // Reports collection - for anonymous user reports
    match /reports/{caseId} {
      // Allow anonymous users to create reports
      allow create: if true;
      
      // Allow anyone to read (for status checking with case ID)
      allow read: if true;
      
      // Only authenticated admins can update/delete reports
      allow update: if isAdmin();
      allow delete: if isAdmin();
    }
    
    // Admin notifications collection - for admin dashboard notifications
    match /admin_notifications/{notificationId} {
      // Only authenticated admins can access notifications
      allow read, write: if isAdmin();
      
      // Allow creation from anywhere (for automatic notification generation)
      allow create: if true;
    }
    
    // Admin users collection - for admin management
    match /admin_users/{userId} {
      // Only authenticated admins can access admin user data
      allow read, write: if isAdmin();
    }
    
    // Admin messages collection - for internal admin communication
    match /admin_messages/{messageId} {
      // Only authenticated admins can access messages
      allow read, write: if isAdmin();
    }
    
    // Deny all other access by default for security
    match /{document=**} {
      allow read, write: if false;
    }
  }
}
